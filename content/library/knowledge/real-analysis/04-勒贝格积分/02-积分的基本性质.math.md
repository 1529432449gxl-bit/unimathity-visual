@chapter 4 | 勒贝格积分
@section 4.2 | 积分的基本性质

@theorem 2.1
设 $f$ 在可测集 $E$ 上可积，$E_1,\ldots,E_n$ 是 $E$ 的两两不交可测子集，则
$$
\begin{aligned}
\int_{\bigcup_{k=1}^{n}E_k}f\,\mathrm dm
=\sum_{k=1}^{n}\int_{E_k}f\,\mathrm dm.
\end{aligned}
$$

@solution 证明
先对简单函数按有限划分验证，再用递增简单函数逼近得到非负函数情形。对一般可积函数分别应用于 $f_+,f_-$ 并相减，因所有积分有限而合法。

@theorem 2.2
若 $f$ 在 $E$ 上可积，则对任意 $\varepsilon>0$，存在 $\delta>0$，使得对每个可测集 $e\subset E$，只要 $me<\delta$，便有
$$
\begin{aligned}
\int_e|f|\,\mathrm dm<\varepsilon.
\end{aligned}
$$

@solution 证明
选 $M$ 使 $\int_{\{|f|>M\}}|f|<\varepsilon/2$，这是可积函数的尾积分趋零性质。令 $\delta=\varepsilon/(2M)$（$M>0$），则 $\int_e|f|\le M m(e)+\int_{|f|>M}|f|<\varepsilon$。

@theorem 2.3
设 $f$ 在可测集 $E$ 上可积，$\{E_k\}_{k\in\mathbf N}$ 是 $E$ 的两两不交可测子集，则
$$
\begin{aligned}
\int_{\bigcup_{k=1}^{\infty}E_k}f\,\mathrm dm
=\sum_{k=1}^{\infty}\int_{E_k}f\,\mathrm dm.
\end{aligned}
$$
右端级数绝对收敛。

基本引理 设 $f(x)$ 是有界可测集 $E$ 上的非负可积函数，$\{f_n(x)\}$ 是满足条件
$$
\begin{aligned}
0\leqslant f_1(x)\leqslant f_2(x)\leqslant\cdots,\qquad \lim_{n\to\infty}f_n(x)=f(x)\ (x\in E)
\end{aligned}
$$
的简单函数列，则
$$
\begin{aligned}
\int_Ef(x)\,\mathrm dm=\lim_{n\to\infty}\int_Ef_n(x)\,\mathrm dm.
\end{aligned}
$$
若不要求 $f(x)$ 在 $E$ 上可积而只要求它有积分，结论仍然正确；若 $f(x)$ 在 $E$ 上不可积，则结论成为 $\lim_{n\to\infty}\int_Ef_n(x)\,\mathrm dm=\infty$。此外，当 $E$ 为无界可测集时基本引理同样成立。

@solution 证明
对非负 $f$ 先应用有限可加性到前 $N$ 个集合，再令 $N\to\infty$，由单调收敛定理得到等式。对可积 $f$ 分别处理正负部分；绝对积分可加性还保证所得级数绝对收敛。

@theorem 2.4
若 $f$ 在 $E$ 上可积，$c$ 是常数，则 $cf$ 在 $E$ 上可积，且
$$
\begin{aligned}
\int_Ecf\,\mathrm dm=c\int_Ef\,\mathrm dm.
\end{aligned}
$$

@solution 证明
对非负函数与 $c\ge0$ 从简单函数积分定义及取上确界得到齐次性。一般实 $f,c$ 则利用正负部分互换或缩放；$\int|cf|=|c|\int|f|<\infty$ 保证可积。

@theorem 2.5
若 $f$、$g$ 在 $E$ 上可积，则 $f+g$ 与 $f-g$ 在 $E$ 上可积，且
$$
\begin{aligned}
\int_E(f+g)\,\mathrm dm&=\int_Ef\,\mathrm dm+\int_Eg\,\mathrm dm,\\
\int_E(f-g)\,\mathrm dm&=\int_Ef\,\mathrm dm-\int_Eg\,\mathrm dm.
\end{aligned}
$$

@solution 证明
$|f\pm g|\le|f|+|g|$ 保证可积。简单函数层面线性成立，用单调简单逼近先得非负函数加法，再对正负部分分组，得到一般和差的线性公式。

@theorem 2.6
若 $f$、$g$ 在 $E$ 上可积，且 $f\leqslant g$ 几乎处处成立，则
$$
\begin{aligned}
\int_Ef\,\mathrm dm\leqslant\int_Eg\,\mathrm dm.
\end{aligned}
$$

@solution 证明
函数 $g-f$ 几乎处处非负且可积，其积分非负，所以 $\int f\le\int g$。再由 $-|f|\le f\le|f|$ 得 $|\int f|\le\int|f|$。

@theorem 2.7
若 $f$ 在 $E$ 上可积，则
$$
\begin{aligned}
\int_E|f|\,\mathrm dm=0
\end{aligned}
$$
的充要条件是 $f=0$ 几乎处处。若 $f=g$ 几乎处处，则 $f$ 与 $g$ 同时可积或同时不可积；可积时有
$$
\begin{aligned}
\int_Ef\,\mathrm dm=\int_Eg\,\mathrm dm.
\end{aligned}
$$

@problem 例题 2.1 | 综合题 | 基础 | |
若 $f$ 在 $\mathbf R$ 上可积，则对任意 $\varepsilon>0$，存在 $\delta>0$，使得任意可测集 $e\subset\mathbf R$ 满足 $me<\delta$ 时都有
$$
\begin{aligned}
\int_e|f|\,\mathrm dm<\varepsilon.
\end{aligned}
$$

@solution 证明
证 设 $f$ 可积，则 $|f|$ 可积。令 $f_N=\min(|f|,N)$，$f_N\uparrow|f|$，由单调收敛定理，$\int f_N\to\int|f|$。故对 $\varepsilon>0$ 有 $N$，使 $\int(|f|-f_N)<\frac\varepsilon2$。取 $\delta=\frac\varepsilon{2N}$，则 $me<\delta$ 时
$$
\int_e|f|=\int_e(|f|-f_N)+\int_ef_N\le\int(|f|-f_N)+N\,me<\frac\varepsilon2+\frac\varepsilon2=\varepsilon .
$$
结论 积分的绝对连续性成立，$\delta$ 与 $e$ 无关。

@lemma 2.1
设 $f(x)$ 是可测集 $E$ 上的可积函数，则对任意的正数 $\varepsilon$，存在 $E$ 上的简单函数 $\varphi(x)$，使
$$
\begin{aligned}
\int_E|f(x)-\varphi(x)|\,\mathrm dm<\varepsilon.
\end{aligned}
$$
若 $f$ 有界，$|f(x)|\leqslant M\ (x\in E)$，则所取的 $\varphi$ 也满足 $|\varphi(x)|\leqslant M\ (x\in E)$。

@lemma 2.2
设 $E$ 是闭区间 $[a,b]$ 中的可测集，则对任何 $\varepsilon>0$，有 $[a,b]$ 上的连续函数 $g(x)$，使
$$
\begin{aligned}
\int_{[a,b]}|\chi_E(x)-g(x)|\,\mathrm dm<\varepsilon.
\end{aligned}
$$

@theorem 2.8
设 $f$ 在闭区间 $[a,b]$ 上可积。对任意 $\varepsilon>0$，存在 $[a,b]$ 上的连续函数 $g$，使得
$$
\begin{aligned}
\int_a^b|f-g|\,\mathrm dm<\varepsilon.
\end{aligned}
$$
若 $|f|\leqslant M$ 几乎处处成立，则还可使 $|g|\leqslant M$ 在 $[a,b]$ 上处处成立。
